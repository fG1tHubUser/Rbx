local gameList = {}
local listLoaded = false

local function loadGameList()
	local success, body = pcall(function() return HttpService:GetAsync(CHAIN_URL, true) end)
	if not success or type(body) ~= "string" then
		warn("[BadgeChain] Failed to load list")
		listLoaded = true
		return
	end

	for line in body:gmatch("[^\r\n]+") do
		line = line:match("^%s*(.-)%s*$")
		if line ~= "" and not line:match("^#") then
			local uniStr, placeStr = line:match("^(%d+)%s*,%s*(%d+)$")
			if uniStr and placeStr then
				table.insert(gameList, {
					universeId = tonumber(uniStr),
					placeId = tonumber(placeStr)
				})
			end
		end
	end

	listLoaded = true
end

task.spawn(loadGameList)

local function fetchBadgesForUniverse(universeId)
	local all = {}
	local cursor = nil
	local pages = 0

	while true do
		pages += 1

		local url = string.format(
			"https://badges.roproxy.com/v1/universes/%d/badges?limit=100&sortOrder=Asc%s",
			universeId,
			cursor and ("&cursor=" .. HttpService:UrlEncode(cursor)) or ""
		)

		local success, response = pcall(function() return HttpService:GetAsync(url, true) end)

		if not success then
			url = string.format("https://badges.roblox.com/v1/universes/%d/badges?limit=100&sortOrder=Asc%s", universeId, cursor and ("&cursor=" .. HttpService:UrlEncode(cursor)) or "")
			success, response = pcall(function() return HttpService:GetAsync(url, true) end)
		end

		if not success then return nil end

		local ok, data = pcall(function() return HttpService:JSONDecode(response) end)
		if not ok or type(data) ~= "table" then return nil end

		if data.data then
			for _, badge in ipairs(data.data) do
				if badge.id and badge.enabled ~= false then
					table.insert(all, { id = badge.id })
				end
			end
		end

		cursor = data.nextPageCursor
		if not cursor or cursor == "" or pages > 40 then break end
		task.wait(0.08)
	end

	return all
end

local function isRateLimitError(err)
	local lower = string.lower(tostring(err))

	for _, keyword in ipairs(RATE_LIMIT_KEYWORDS) do
		if string.find(lower, keyword, 1, true) then return true end
	end

	return false
end

local function setPhase(gui, statusStr, infoStr)
	if gui and gui.statusText and gui.currentStateInfo then
		gui.statusText.Text = statusStr
		gui.currentStateInfo.Text = infoStr
	end
end

local function updateProgress(gui, owned, total)
	if not gui then return end

	local alpha = (total > 0) and math.clamp(owned / total, 0, 1) or 0
	TweenService:Create(gui.barFill, TWEEN_BAR, {Size = UDim2.new(MAX_BAR_WIDTH * alpha, 0, BAR_HEIGHT, 0)}):Play()
	gui.awardedOfTotal.Text = string.format("%d / %d", owned, total)
	gui.remainingBadges.Text = string.format("%d Remaining", total - owned)
end

local function processPlayer(player)
	while not listLoaded do task.wait(0.5) end

	local gui = createGui(player)
	if not gui then return end
	setPhase(gui, "Waiting...", "Loading game data...")

	local currentUniverse = game.GameId
	local currentPlace = game.PlaceId
	local currentIndex = nil

	for i, entry in ipairs(gameList) do
		if entry.universeId == currentUniverse then
			currentIndex = i
			break
		end
	end

	local nextPlaceId = (currentIndex and currentIndex < #gameList) and gameList[currentIndex + 1].placeId or nil
	gui.gameTracker.Text = string.format("Game %d of %d", currentIndex or 1, #gameList > 0 and #gameList or 1)

	local dsKey = "User_" .. player.UserId .. "_Universe_" .. currentUniverse
	local alreadyAwarded = {}

	-- Recover from TeleportData (short-term cache)
	local joinData = player:GetJoinData()
	local teleportData = joinData and joinData.TeleportData or {}

	if teleportData.awardedIds then
		for _, id in ipairs(teleportData.awardedIds) do
			alreadyAwarded[tostring(id)] = true
		end
	end

	-- LOAD PERMANENT DATASTORE CACHE (long-term cache)
	local dsSuccess, dsData = pcall(function() return BadgeCache:GetAsync(dsKey) end)
	if dsSuccess and type(dsData) == "table" then
		for _, id in ipairs(dsData) do
			alreadyAwarded[tostring(id)] = true
		end
	end

	-- Function to save current progress to permanent cache before teleports
	local function saveCache()
		local toSave = {}
		for id in pairs(alreadyAwarded) do
			table.insert(toSave, tonumber(id))
		end
		pcall(function() BadgeCache:SetAsync(dsKey, toSave) end)
	end

	setPhase(gui, "Loading badges...", "Loading chain data...")
	local badges = fetchBadgesForUniverse(currentUniverse)

	-- Automatic rejoin if badge loading fails
	if not badges then
		for seconds = 5, 1, -1 do
			if not player.Parent then return end

			setPhase(gui, "Error", string.format("Failed to load badges. Rejoining in %d...", seconds))
			task.wait(1)
		end

		if not player.Parent then return end

		setPhase(gui, "Error", "Failed to load badges. Rejoining...")

		while player.Parent do
			pcall(function() TeleportService:TeleportAsync(currentPlace, {player}) end)
			task.wait(2.5)
		end

		return
	end

	local toAward = {}
	local ownedCount = 0
	local total = #badges

	setPhase(gui, "Checking badges...", "Syncing inventory...")

	for _, badge in ipairs(badges) do
		-- If it's in the permanent cache or teleport data, skip checking entirely!
		if alreadyAwarded[tostring(badge.id)] then
			ownedCount += 1
		else
			local has = false
			local ok, result = pcall(function() return BadgeService:UserHasBadgeAsync(player.UserId, badge.id) end)

			if ok and result then
				has = true
				ownedCount += 1
				alreadyAwarded[tostring(badge.id)] = true
			end

			if not has then table.insert(toAward, badge) end
		end
	end

	updateProgress(gui, ownedCount, total)
	task.wait(1.5)

	local AWARDS_PER_SERVER = 85
	local awardsThisServer = 0

	-- Award Loop (Only iterates through badges the player does NOT own)
	for _, badge in ipairs(toAward) do
		if not player.Parent then return end

		local success, result = pcall(function() return BadgeService:AwardBadgeAsync(player.UserId, badge.id) end)

		if success and result == true then
			ownedCount += 1
			awardsThisServer += 1
			alreadyAwarded[tostring(badge.id)] = true
			setPhase(gui, "Awarding badges...", "Awarded Badge: " .. tostring(badge.id))
			updateProgress(gui, ownedCount, total)

			if awardsThisServer >= AWARDS_PER_SERVER then
				setPhase(gui, "Switching servers...", "Rate limit reached. Moving to new server...")
				task.wait(1.5)

				saveCache() -- Save to DataStore before leaving

				local teleportOptions = Instance.new("TeleportOptions")
				teleportOptions.ShouldReserveServer = true
				local saveData = { awardedIds = {}, fromChain = true }

				for id in pairs(alreadyAwarded) do
					table.insert(saveData.awardedIds, tonumber(id))
				end

				teleportOptions:SetTeleportData(saveData)

				while player.Parent do
					pcall(function() TeleportService:TeleportAsync(currentPlace, {player}, teleportOptions) end)
					task.wait(2.5)
				end

				return
			end

			task.wait(AWARD_DELAY)

		elseif success and result == false then
			-- Badge was already owned natively but missed the check. Skip instantly.
			ownedCount += 1
			alreadyAwarded[tostring(badge.id)] = true
			updateProgress(gui, ownedCount, total)
		else
			-- Handle errors without triggering fake server jumps
			if isRateLimitError(result) then
				warn("[BadgeChain] Rate limited by Roblox API. Pausing...")
			else
				warn("[BadgeChain] Error awarding badge " .. tostring(badge.id) .. ": " .. tostring(result))
			end
		end
	end

	updateProgress(gui, total, total)
	saveCache() -- Save to DataStore once all badges in this game are cleared

	-- Play completion sound
	local completeSound = Instance.new("Sound")
	completeSound.SoundId = "rbxassetid://107978604605265"
	completeSound.Volume = 0.69
	completeSound.PlaybackSpeed = 0.5
	completeSound.Parent = gui.screen
	completeSound:Play()

	if nextPlaceId then
		setPhase(gui, "Awarded all badges in this game!", "Teleporting to the next...")
		task.wait(2)

		while player.Parent do
			pcall(function() TeleportService:Teleport(nextPlaceId, player, { fromChain = true }) end)
			task.wait(2.5)
		end
	else
		setPhase(gui, "Come back tomorrow for more badges!", "Chain complete!")
		gui.remainingBadges.Text = "All badges awarded in every game!"
	end
end

Players.PlayerAdded:Connect(function(player) task.spawn(processPlayer, player) end)

for _, player in ipairs(Players:GetPlayers()) do task.spawn(processPlayer, player) end
