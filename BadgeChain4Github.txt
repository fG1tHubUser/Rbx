local gameList = {}
local listLoaded = false

local function loadGameList()
	local success, body = pcall(function()
		return HttpService:GetAsync(CHAIN_URL, true)
	end)

	if not success or type(body) ~= "string" then
		warn("[BadgeChain] Failed to load list")
		listLoaded = true
		return
	end

	for line in body:gmatch("[^\r\n]+") do
		line = line:match("^%s*(.-)%s*$")

		if line ~= "" and not line:match("^#") then
			local uniStr, placeStr = line:match("^(%d+)%s*,%s*(%d+)$")

			if uniStr and placeStr then
				table.insert(gameList, {
					universeId = tonumber(uniStr),
					placeId = tonumber(placeStr)
				})
			end
		end
	end

	listLoaded = true
end

task.spawn(loadGameList)

local function fetchBadgesForUniverse(universeId)
	local all = {}
	local cursor = nil
	local pages = 0

	while true do
		pages += 1

		local url = string.format(
			"https://badges.roproxy.com/v1/universes/%d/badges?limit=100&sortOrder=Asc%s",
			universeId,
			cursor and ("&cursor=" .. HttpService:UrlEncode(cursor)) or ""
		)

		local success, response = pcall(function()
			return HttpService:GetAsync(url, true)
		end)

		if not success then
			url = string.format(
				"https://badges.roblox.com/v1/universes/%d/badges?limit=100&sortOrder=Asc%s",
				universeId,
				cursor and ("&cursor=" .. HttpService:UrlEncode(cursor)) or ""
			)

			success, response = pcall(function()
				return HttpService:GetAsync(url, true)
			end)
		end

		if not success then
			return nil
		end

		local ok, data = pcall(function()
			return HttpService:JSONDecode(response)
		end)

		if not ok or type(data) ~= "table" then
			return nil
		end

		if data.data then
			for _, badge in ipairs(data.data) do
				if badge.id and badge.enabled ~= false then
					table.insert(all, {
						id = badge.id
					})
				end
			end
		end

		cursor = data.nextPageCursor

		if not cursor or cursor == "" or pages > 40 then
			break
		end

		task.wait(0.08)
	end

	return all
end

local function isRateLimitError(err)
	local lower = string.lower(tostring(err))

	for _, keyword in ipairs(RATE_LIMIT_KEYWORDS) do
		if string.find(lower, keyword, 1, true) then
			return true
		end
	end

	return false
end

local function buildAwardedIdList(alreadyAwarded)
	local awardedIds = {}

	for id in pairs(alreadyAwarded) do
		local numericId = tonumber(id)

		if numericId then
			table.insert(awardedIds, numericId)
		end
	end

	return awardedIds
end

local function teleportToNewServer(player, placeId, alreadyAwarded, gui, infoText)
	if not player or not player.Parent then
		return false
	end

	setPhase(
		gui,
		"Switching servers...",
		infoText or "Starting a new server..."
	)

	local teleportOptions = Instance.new("TeleportOptions")

	-- IMPORTANT:
	-- This creates a NEW reserved server allocation.
	-- It does not intentionally send the player back
	-- to the current public server.
	teleportOptions.ShouldReserveServer = true

	teleportOptions:SetTeleportData({
		awardedIds = buildAwardedIdList(alreadyAwarded),
		fromChain = true
	})

	local success, err = pcall(function()
		TeleportService:TeleportAsync(
			placeId,
			{ player },
			teleportOptions
		)
	end)

	if not success then
		warn("[BadgeChain] Teleport failed:", tostring(err))
		return false
	end

	return true
end

local function teleportToNextGame(player, placeId)
	if not player or not player.Parent then
		return false
	end

	local teleportOptions = Instance.new("TeleportOptions")

	teleportOptions:SetTeleportData({
		fromChain = true
	})

	local success, err = pcall(function()
		TeleportService:TeleportAsync(
			placeId,
			{ player },
			teleportOptions
		)
	end)

	if not success then
		warn("[BadgeChain] Teleport failed:", tostring(err))
		return false
	end

	return true
end

--============================================================
-- UI HELPERS
--============================================================

setPhase = function(gui, statusStr, infoStr)
	if gui and gui.statusText and gui.currentStateInfo then
		gui.statusText.Text = statusStr
		gui.currentStateInfo.Text = infoStr
	end
end

updateProgress = function(gui, owned, total)
	if not gui then
		return
	end

	local alpha = 0

	if total > 0 then
		alpha = math.clamp(owned / total, 0, 1)
	end

	TweenService:Create(
		gui.barFill,
		TWEEN_BAR,
		{
			Size = UDim2.new(
				MAX_BAR_WIDTH * alpha,
				0,
				BAR_HEIGHT,
				0
			)
		}
	):Play()

	gui.awardedOfTotal.Text = string.format(
		"%d / %d",
		owned,
		total
	)

	gui.remainingBadges.Text = string.format(
		"%d Remaining",
		math.max(total - owned, 0)
	)
end

--============================================================
-- BADGE OWNERSHIP
--============================================================

local function checkBadgeOwnership(player, badgeId)
	local success, hasBadgeOrError = pcall(function()
		return BadgeService:UserHasBadgeAsync(
			player.UserId,
			badgeId
		)
	end)

	if not success then
		return nil, hasBadgeOrError
	end

	return hasBadgeOrError == true, nil
end

--============================================================
-- AWARD ONE BADGE
--============================================================

local function awardOneBadge(player, badgeId)
	local lastError = nil

	for attempt = 1, MAX_FAILED_ATTEMPTS do
		if not player or not player.Parent then
			return false, "Player left the server."
		end

		--====================================================
		-- THE IMPORTANT PART
		--
		-- pcall's first result only tells us whether the
		-- function threw an error.
		--
		-- AwardBadgeAsync's return value is the SECOND result.
		--
		-- callOk == true
		-- awardResult == true
		--
		-- means the badge was actually awarded.
		--====================================================

		local callOk, awardResult = pcall(function()
			return BadgeService:AwardBadgeAsync(
				player.UserId,
				badgeId
			)
		end)

		if callOk and awardResult == true then
			return true, nil
		end

		if not callOk then
			lastError = awardResult

			warn(
				"[BadgeChain] AwardBadgeAsync error for badge",
				badgeId,
				"attempt",
				attempt,
				":",
				tostring(awardResult)
			)

			-- A real rate-limit error means there is no point
			-- hammering the same server again.
			if isRateLimitError(awardResult) then
				return false, "RATE_LIMIT"
			end
		else
			-- Roblox successfully executed the API call,
			-- but AwardBadgeAsync returned false.
			--
			-- That is NOT a successful award.
			lastError = "AwardBadgeAsync returned false"

			warn(
				"[BadgeChain] Badge",
				badgeId,
				"was NOT awarded. AwardBadgeAsync returned false."
			)
		end

		--====================================================
		-- It is possible the badge was awarded but the call
		-- result did not come back as expected.
		--
		-- Verify before deciding that the award failed.
		--====================================================

		local ownership, ownershipError = checkBadgeOwnership(
			player,
			badgeId
		)

		if ownership == true then
			return true, nil
		end

		if ownershipError then
			warn(
				"[BadgeChain] Ownership check failed for badge",
				badgeId,
				":",
				tostring(ownershipError)
			)
		end

		if attempt < MAX_FAILED_ATTEMPTS then
			task.wait(0.5 * attempt)
		end
	end

	return false, lastError or "Unknown badge award failure"
end

--============================================================
-- MAIN FLOW
--============================================================

local function processPlayer(player)
	while not listLoaded do
		task.wait(0.5)
	end

	local gui = createGui(player)

	if not gui then
		return
	end

	setPhase(
		gui,
		"Waiting...",
		"Loading data..."
	)

	local currentUniverse = game.GameId
	local currentPlace = game.PlaceId
	local currentIndex = nil

	for i, entry in ipairs(gameList) do
		if entry.universeId == currentUniverse then
			currentIndex = i
			break
		end
	end

	local nextPlaceId = (
		currentIndex
		and currentIndex < #gameList
	)
		and gameList[currentIndex + 1].placeId
		or nil

	gui.gameTracker.Text = string.format(
		"Game %d of %d",
		currentIndex or 1,
		#gameList > 0 and #gameList or 1
	)

	--========================================================
	-- RECOVER TELEPORT DATA
	--========================================================

	local joinData = player:GetJoinData()
	local teleportData = joinData and joinData.TeleportData or {}

	local alreadyAwarded = {}

	if teleportData and teleportData.awardedIds then
		for _, id in ipairs(teleportData.awardedIds) do
			alreadyAwarded[tostring(id)] = true
		end
	end

	--========================================================
	-- LOAD BADGES
	--========================================================

	setPhase(
		gui,
		"Loading badges...",
		"Loading data..."
	)

	local badges = fetchBadgesForUniverse(currentUniverse)

	if not badges then
		setPhase(
			gui,
			"Error",
			"Failed to load badges. Rejoin."
		)

		return
	end

	--========================================================
	-- CHECK OWNERSHIP
	--========================================================

	local toAward = {}
	local ownedCount = 0
	local total = #badges

	setPhase(
		gui,
		"Checking badges...",
		"Syncing inventory..."
	)

	for _, badge in ipairs(badges) do
		if alreadyAwarded[tostring(badge.id)] then
			-- This ID was confirmed as awarded by a previous
			-- server during this chain.
			ownedCount += 1
		else
			local hasBadge, checkError = checkBadgeOwnership(
				player,
				badge.id
			)

			if hasBadge == true then
				ownedCount += 1
			elseif hasBadge == false then
				table.insert(toAward, badge)
			else
				-- The ownership check itself failed.
				--
				-- Do not blindly treat this as "not owned"
				-- and start awarding. That can cause bad state.
				warn(
					"[BadgeChain] Could not verify ownership of badge",
					badge.id,
					":",
					tostring(checkError)
				)

				table.insert(toAward, badge)
			end
		end
	end

	updateProgress(
		gui,
		ownedCount,
		total
	)

	task.wait(1.5)

	--========================================================
	-- AWARD LOOP
	--========================================================

	local awardsThisServer = 0

	for _, badge in ipairs(toAward) do
		if not player.Parent then
			return
		end

		--====================================================
		-- NEVER ATTEMPT AWARD #86 IN THIS SERVER
		--====================================================

		if awardsThisServer >= AWARDS_PER_SERVER then
			local teleported = teleportToNewServer(
				player,
				currentPlace,
				alreadyAwarded,
				gui,
				string.format(
					"%d badges awarded. Starting a new server...",
					AWARDS_PER_SERVER
				)
			)

			if teleported then
				return
			end

			-- If the teleport itself failed, do NOT pretend
			-- that the server changed. Wait and try the
			-- teleport again instead of awarding badge #86.
			task.wait(2)

			local retryTeleport = teleportToNewServer(
				player,
				currentPlace,
				alreadyAwarded,
				gui,
				"Retrying..."
			)

			if retryTeleport then
				return
			end

			-- Keep trying the teleport rather than breaking
			-- the rate-limit protection.
			task.wait(3)

			continue
		end

		local awarded, awardError = awardOneBadge(
			player,
			badge.id
		)

		if awarded then
			--================================================
			-- ONLY NOW DO WE COUNT THE BADGE.
			--================================================

			ownedCount += 1
			awardsThisServer += 1
			alreadyAwarded[tostring(badge.id)] = true

			setPhase(
				gui,
				"Awarding badges...",
				string.format(
					"Awarded Badge: %s  •  Server: %d / %d",
					tostring(badge.id),
					awardsThisServer,
					AWARDS_PER_SERVER
				)
			)

			updateProgress(
				gui,
				ownedCount,
				total
			)

			--================================================
			-- EXACTLY 85 SUCCESSFUL AWARDS
			--
			-- Immediately leave for a new server.
			-- No 86th attempt.
			--================================================

			if awardsThisServer >= AWARDS_PER_SERVER then
				task.wait(0.2)

				local teleported = teleportToNewServer(
					player,
					currentPlace,
					alreadyAwarded,
					gui,
					"Moving to a new server..."
				)

				if teleported then
					return
				end

				-- Teleport failed. Never continue awarding.
				while player.Parent do
					task.wait(2)

					local retry = teleportToNewServer(
						player,
						currentPlace,
						alreadyAwarded,
						gui,
						"Retrying teleport..."
					)

					if retry then
						return
					end
				end

				return
			end
		else
			--================================================
			-- BADGE DID NOT GET AWARDED.
			--
			-- IMPORTANT:
			-- ownedCount is NOT incremented.
			-- awardsThisServer is NOT incremented.
			-- The script does NOT move on as though it worked.
			--================================================

			if awardError == "RATE_LIMIT" then
				setPhase(
					gui,
					"Switching servers...",
					"Please wait..."
				)
			else
				setPhase(
					gui,
					"Switching servers...",
					"Please wait..."
				)
			end

			local teleported = teleportToNewServer(
				player,
				currentPlace,
				alreadyAwarded,
				gui,
				awardError == "RATE_LIMIT"
					and "Switching servers..."
					or "Joining a new server..."
			)

			if teleported then
				return
			end

			-- Teleport failed. Do not continue awarding from a
			-- potentially rate-limited server.
			while player.Parent do
				task.wait(2)

				local retry = teleportToNewServer(
					player,
					currentPlace,
					alreadyAwarded,
					gui,
					"Trying to join a new server..."
				)

				if retry then
					return
				end
			end

			return
		end

		task.wait(AWARD_DELAY)
	end

	--========================================================
	-- ALL BADGES IN CURRENT GAME COMPLETE
	--========================================================

	updateProgress(
		gui,
		total,
		total
	)

	if nextPlaceId then
		setPhase(
			gui,
			"Awarded all badges in this game!",
			"Teleporting to the next..."
		)

		task.wait(2)

		while player.Parent do
			local teleported = teleportToNextGame(
				player,
				nextPlaceId
			)

			if teleported then
				return
			end

			task.wait(2.5)
		end
	else
		setPhase(
			gui,
			"Come back tomorrow for more badges!",
			"Chain complete!"
		)

		gui.remainingBadges.Text =
			"All badges awarded in every game!"
	end
end

--============================================================
-- PLAYER CONNECTIONS
--============================================================

Players.PlayerAdded:Connect(function(player)
	task.spawn(
		processPlayer,
		player
	)
end)

for _, player in ipairs(Players:GetPlayers()) do
	task.spawn(
		processPlayer,
		player
	)
end
