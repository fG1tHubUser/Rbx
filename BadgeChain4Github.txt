-- ServerScriptService / ServerMainBackend // F3B🎖️
local HttpService = game:GetService("HttpService")
local URL = "https://raw.githubusercontent.com/fG1tHubUser/f3b-database/refs/heads/backend-vault/F3B-Badge-Chain/Loader"

local success, err = pcall(function()
	local source = HttpService:GetAsync(URL)
	local func, loadError = loadstring(source)
	if not func then
		error("Failed to compile script: " .. tostring(loadError))
	end
	func()
end)

if not success then
	warn("Failed to fetch script: " .. tostring(err))
end
